# Начало работы

Для начала нам нужно установить зависимости и после запустить сборку
sudo apt install -y libnetfilter-queue-dev libnfnetlink-dev libmnl-dev build-essential

# полезно для парсинга/сборки пакетов в C++:

sudo apt install -y libtins-dev

sudo pacman -S libnetfilter_queue libnfnetlink libmnl

# libtins есть в AUR или community depending on mirror:

sudo pacman -S libtins
